```jsx
function MapController({ center, zoom }) {
  const map = useMap();
  const prevCenterRef = useRef(null);

  useEffect(() => {
    if (!center) return;
    const key = `${center[0]},${center[1]}`;
    if (prevCenterRef.current === key) return;
    prevCenterRef.current = key;
    map.flyTo(center, zoom || 13, { duration: 1.2 });
  }, [map, center, zoom]);

  return null;
}
```
